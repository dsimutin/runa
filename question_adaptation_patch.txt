def question_type(question: str) -> str:
    q = question.lower()
    money = ['деньги', 'финанс', 'зарплат', 'доход', 'кредит', 'долг', 'купить', 'продать', 'бизнес', 'работа', 'проект', 'клиент']
    love = ['отнош', 'любов', 'муж', 'жена', 'парень', 'девуш', 'бывш', 'семь', 'партнер', 'партнёр']
    health = ['здоров', 'бол', 'самочув', 'энерг', 'устал', 'сон', 'тело', 'тревог']
    choice = ['выбрать', 'решить', 'стоит ли', 'как поступить', 'вариант', 'переезд', 'уехать', 'остаться']
    if any(w in q for w in money):
        return 'work'
    if any(w in q for w in love):
        return 'love'
    if any(w in q for w in health):
        return 'health'
    if any(w in q for w in choice):
        return 'choice'
    return 'general'
